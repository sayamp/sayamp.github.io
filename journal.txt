/* ════════════════════════════════════════════════════════════════════════

const KEY = "journal:entries";
const Store = {
  async load() {
    try {
      if (window.storage) {
        try { const r = await window.storage.get(KEY); return r ? JSON.parse(r.value) : null; }
        catch { return null; }
      }
      const raw = localStorage.getItem(KEY);
      return raw ? JSON.parse(raw) : null;
    } catch { return null; }
  },
  async save(entries) {
    const val = JSON.stringify(entries);
    try {
      if (window.storage) { await window.storage.set(KEY, val); return; }
      localStorage.setItem(KEY, val);
    } catch (e) { console.error("save failed", e); toast("Couldn't save — storage unavailable"); }
  }
};

const fmtDate = (ts) =>

const esc = (s) => s.replace(/[&<>"']/g, c =>
  ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));

// tiny, safe inline formatting: *italic* and **bold**, paragraphs on blank lines
function toProse(text) {
  return esc(text)
    .split(/\n{2,}/)
    .map(p => {
      const inline = p
        .replace(/\*\*([^*]+)\*\*/g, "<strong>$1</strong>")
        .replace(/\*([^*]+)\*/g, "<em>$1</em>")
        .replace(/\n/g, "<br>");
      return `<p>${inline}</p>`;
    })
    .join("");
}

const excerpt = (text, n = 180) => {
  const flat = text.replace(/[*#_>]/g, "").replace(/\s+/g, " ").trim();
  return flat.length > n ? flat.slice(0, n).trimEnd() + "…" : flat;
};

  return `
    <section class="about view">
      <div class="wrap"><div class="about-inner">
        <span class="eyebrow">${esc(SITE.eyebrow)}</span>
        <h2>${esc(SITE.name)}</h2>
        ${SITE.about.map(p => `<p>${esc(p)}</p>`).join("")}
      </div></div>
    </section>`;
}

/* ── render + wire events ─────────────────────────────────────────────── */
function render() {
  let html;
  if (route.name === "reader") html = viewReader(route.id);
  else if (route.name === "compose") html = viewCompose(route.id);
  else if (route.name === "about") html = viewAbout();
  else html = viewHome();
  app.innerHTML = html;
  paintChrome();
  wire();
}

function wire() {
  app.querySelectorAll("[data-go]").forEach(el =>
    el.addEventListener("click", () => go(el.dataset.go, el.dataset.id || null)));
  app.querySelectorAll("[data-open]").forEach(el =>
    el.addEventListener("click", () => go("reader", el.dataset.open)));
  app.querySelectorAll("[data-edit]").forEach(el =>
    el.addEventListener("click", () => go("compose", el.dataset.edit)));
  app.querySelectorAll("[data-del]").forEach(el =>
    el.addEventListener("click", () => del(el.dataset.del)));

  const save = $("#save");
  if (save) {
    const title = $("#f-title"), body = $("#f-body");
    const auto = () => { body.style.height = "auto"; body.style.height = Math.max(body.scrollHeight, window.innerHeight * 0.46) + "px"; };
    auto(); body.addEventListener("input", auto);
    save.addEventListener("click", () => commit(route.id));
    // ⌘/Ctrl + Enter to save
    [title, body].forEach(f => f.addEventListener("keydown", ev => {
      if ((ev.metaKey || ev.ctrlKey) && ev.key === "Enter") { ev.preventDefault(); commit(route.id); }
    }));
    (title.value ? body : title).focus();
  }
}

/* ── mutations ────────────────────────────────────────────────────────── */
async function commit(id) {
  const title = $("#f-title").value.trim();
  const body = $("#f-body").value.trim();
  if (!title && !body) { toast("Write something first"); return; }
  $("#save-state").textContent = "Saving…";

  if (id) {
    const e = entries.find(x => x.id === id);
    e.title = title; e.body = body; e.edited = Date.now();
  } else {
    id = uid();
    entries.push({ id, title, body, date: Date.now() });
  }
  await Store.save(entries);
  toast("Kept");
  go("reader", id);
}

async function del(id) {
  if (!confirm("Delete this entry for good?")) return;
  entries = entries.filter(x => x.id !== id);
  await Store.save(entries);
  toast("Deleted");
  go("home");
}

/* ── export / import (backups) ────────────────────────────────────────── */
$("#export").addEventListener("click", () => {
  const blob = new Blob([JSON.stringify(entries, null, 2)], { type: "application/json" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url; a.download = (SITE.name || "journal").toLowerCase().replace(/\s+/g, "-") + "-backup.json";
  a.click(); URL.revokeObjectURL(url);
  toast("Backup downloaded");
});
$("#import").addEventListener("click", () => $("#file-input").click());
$("#file-input").addEventListener("change", (ev) => {
  const file = ev.target.files[0]; if (!file) return;
  const reader = new FileReader();
  reader.onload = async () => {
    try {
      const data = JSON.parse(reader.result);
      if (!Array.isArray(data)) throw new Error("bad");
      const ids = new Set(entries.map(e => e.id));
      data.forEach(e => { if (!ids.has(e.id)) entries.push(e); });
      await Store.save(entries); toast("Entries imported"); go("home");
    } catch { toast("That file didn't look right"); }
  };
  reader.readAsText(file);
});

/* ── boot ─────────────────────────────────────────────────────────────── */
(async function init() {
  const loaded = await Store.load();
  entries = loaded && loaded.length ? loaded : SEED.slice();
  if (!loaded) await Store.save(entries);
  render();
})();
</script>
